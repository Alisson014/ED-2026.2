# Variáveis e memória:
uma variável ocupa o tamanho referente ao seu tipo na memória;
Valor é o que está armazenado e endereço é a localização na memória
    printf("%p" ,   (void *)&var) para imprimir o endereço na memória;

# Operações algébricas: igual python;



Operador ++:
 ++x; incrementa primeiro e retorna o valor dempois;
 x++; retorna o valor primeiro e incrementa depois;

 int x = 5;
 int resultado = x++;     // resultado == 5 e x == 6

 x = 5
 resultado = ++x;         // resultado == 6 e x == 6

# Recursividade:
 int fatorial(int n) {
    // caso base
    if (n == 0){
        return 1;
    }

    return n * fatorial(n - 1);
 }
